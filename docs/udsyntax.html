udsyntax

udsyntax: extract simple syntax graphs from spaCy Universal Dependencies parses.

Typical usage::

from udsyntax import load_latin, SyntaxGraph

nlp = load_latin()
doc = nlp("Gallia est omnis divisa in partes tres.")
graph = SyntaxGraph.from_doc(doc)
df = graph.to_polars()

 31    """A dependency parse represented as syntax-graph nodes and edges."""

 37    @classmethod
 38    def from_doc(cls, doc, *, urn: str | None = None) -> "SyntaxGraph":
 39        """Build a ``SyntaxGraph`` from a parsed spaCy ``Doc``.
 40
 41        ``doc`` must come from a pipeline that has run a dependency
 42        parser (its tokens need ``.dep_``, ``.head``, and ``.morph``).
 43        """
 44        nodes: list[SyntaxNode] = []
 45        edges: list[SyntaxEdge] = []
 46        has_sents = doc.has_annotation("SENT_START")
 47        for tok in doc:
 48            sent_id = tok.sent.start if has_sents else None
 49            nodes.append(
 50                SyntaxNode(
 51                    id=tok.i,
 52                    text=tok.text,
 53                    lemma=tok.lemma_,
 54                    pos=tok.pos_,
 55                    relation=tok.dep_,
 56                    head_id=tok.head.i,
 57                    morph=tok.morph.to_dict(),
 58                    sent_id=sent_id,
 59                )
 60            )
 61            if tok.dep_ != "ROOT":
 62                edges.append(SyntaxEdge(src=tok.head.i, target=tok.i, relation=tok.dep_))
 63        return cls(nodes=nodes, edges=edges, urn=urn)
 64
 65    def node(self, node_id: int) -> SyntaxNode:
 66        for n in self.nodes:
 67            if n.id == node_id:
 68                return n
 69        raise KeyError(node_id)
 70
 71    def root_nodes(self) -> list[SyntaxNode]:
 72        return [n for n in self.nodes if n.is_root]
 73
 74    def children_of(self, node_id: int) -> list[SyntaxNode]:
 75        child_ids = {e.target for e in self.edges if e.src == node_id}
 76        return [n for n in self.nodes if n.id in child_ids]
 77
 78    def verbal_units(self) -> list[VerbalUnit]:
 79        """Finite verbs in this graph, as ``VerbalUnit`` records.
 80
 81        See :class:`udsyntax.models.VerbalUnit` for the current
 82        limitation on ``depth``.
 83        """
 84        units = []
 85        for i, n in enumerate(node for node in self.nodes if node.morph.get("VerbForm") == "Fin"):
 86            depth = 1 if n.is_root else None
 87            units.append(VerbalUnit(id=i, verb_token_id=n.id, verb_text=n.text, depth=depth))
 88        return units

 90    def to_polars(self):
 91        """Render this graph's nodes as a polars ``DataFrame``, one row per token."""
 92        import polars as pl
 93
 94        rows = []
 95        for n in self.nodes:
 96            row = {
 97                "id": n.id,
 98                "text": n.text,
 99                "lemma": n.lemma,
100                "pos": n.pos,
101                "syntax": n.relation,
102                "parentid": n.head_id,
103                "sent_id": n.sent_id,
104                **n.morph,
105            }
106            rows.append(row)
107        return pl.DataFrame(rows)

109    def to_networkx(self):
110        """Render this graph as a ``networkx.DiGraph``, with token data as node attrs.
111
112        Requires the optional ``networkx`` dependency (``pip install
113        "udsyntax[graph]"``).
114        """
115        import networkx as nx

117        g = nx.DiGraph()
118        for n in self.nodes:
119            g.add_node(n.id, text=n.text, lemma=n.lemma, pos=n.pos, relation=n.relation, **n.morph)
120        for e in self.edges:
121            g.add_edge(e.src, e.target, relation=e.relation)
122        return g

124    def to_mermaid(self, orientation: str = "BT") -> str:
125        """Render this graph as a Mermaid flowchart definition.
126
127        Each node is rendered as ``n<id>["text (id:pos)"]`` and each
128        dependency edge as ``n<head> -->|relation| n<dependent>``, so the
129        arrow direction always runs from a token to its dependent,
130        regardless of ``orientation``.
131
132        Parameters
133        ----------
134        orientation:
135            The direction keyword written after ``graph`` in the header
136            line -- one of ``"TB"``, ``"TD"``, ``"BT"``, ``"RL"``, or
137            ``"LR"`` (see the Mermaid flowchart docs). Defaults to
138            ``"BT"`` (bottom-to-top), which reads like a traditional
139            syntax tree with the root at the top.
140
141        No external dependency is required -- this returns plain text
142        that can be pasted into any Mermaid-aware renderer (GitHub,
143        Claude/Cowork artifacts, the Mermaid Live Editor, etc.).
144        """
145        if orientation not in MERMAID_ORIENTATIONS:
146            raise ValueError(
147                f"orientation must be one of {sorted(MERMAID_ORIENTATIONS)}, got {orientation!r}"
148            )
149
150        lines = [f"graph {orientation}"]
151        for n in self.nodes:
152            label = _mermaid_escape(f"{n.text} ({n.id}:{n.pos})")
153            lines.append(f'    n{n.id}["{label}"]')
154        for e in self.edges:
155            label = _mermaid_escape(e.relation)
156            lines.append(f"    n{e.src} -->|{label}| n{e.target}")
157        return "\n".join(lines)
158
159    def to_dot(self) -> str:
160        """Render this graph as a Graphviz DOT ``digraph`` definition.
161
162        Each node is rendered as ``n<id> [label="text (id:pos)"];`` and
163        each dependency edge as ``n<head> -> n<dependent>
164        [label="relation"];``. The digraph is named after ``self.urn``
165        when present, otherwise ``"SyntaxGraph"``.
166
167        No external dependency is required -- this returns plain DOT
168        source text; feed it to the ``dot`` command line tool or the
169        ``graphviz`` Python package to render an image.
170        """
171        graph_name = _dot_escape(self.urn if self.urn else "SyntaxGraph")
172        lines = [f'digraph "{graph_name}" {{']
173        for n in self.nodes:
174            label = _dot_escape(f"{n.text} ({n.id}:{n.pos})")
175            lines.append(f'    n{n.id} [label="{label}"];')
176        for e in self.edges:
177            label = _dot_escape(e.relation)
178            lines.append(f"    n{e.src} -> n{e.target} [label=\"{label}\"];")
179        lines.append("}")
180        return "\n".join(lines)

def corpus_to_polars(graphs: Iterable[SyntaxGraph]): 
183def corpus_to_polars(graphs: Iterable[SyntaxGraph]):
184    """Concatenate several ``SyntaxGraph``\\ s into one polars ``DataFrame``.
185
186    Adds a ``doc_index`` column (position of each graph in ``graphs``).
187    When a graph's ``urn`` is a parseable CTS URN, ``group`` / ``work`` /
188    ``version`` / ``passage`` columns are added alongside it.
189    """
190    import polars as pl
191
192    frames = []
193    for i, g in enumerate(graphs):
194        df = g.to_polars().with_columns(pl.lit(i).alias("doc_index"))
195        if g.urn:
196            try:
197                parsed = parse_cts_urn(g.urn)
198            except ValueError:
199                df = df.with_columns(pl.lit(g.urn).alias("urn"))
200            else:
201                df = df.with_columns(
202                    pl.lit(g.urn).alias("urn"),
203                    pl.lit(parsed.group).alias("group"),
204                    pl.lit(parsed.work).alias("work"),
205                    pl.lit(parsed.version).alias("version"),
206                    pl.lit(parsed.passage).alias("passage"),
207                )
208        frames.append(df)
209    return pl.concat(frames, how="diagonal_relaxed") if frames else pl.DataFrame()

def read_cex( path, *, urn_filter: str | None = None, encoding: str = 'utf-8') -> list[tuple[str, str]]: 
14def read_cex(
15    path,
16    *,
17    urn_filter: str | None = None,
18    encoding: str = "utf-8",
19) -> list[tuple[str, str]]:
20    """Read a CEX file of ``urn|text`` lines.
21
22    Parameters
23    ----------
24    path:
25        Path to a ``.cex`` file.
26    urn_filter:
27        If given, keep only lines whose URN contains this substring
28        (e.g. ``"vulgate"``, ``"septuagint"``, ``"targum_latin"``).
29    encoding:
30        Text encoding to use when reading the file.
31
32    Returns
33    -------
34    A list of ``(urn, text)`` tuples, in file order. Blank lines are
35    skipped.
36    """
37    path = Path(path)
38    pairs: list[tuple[str, str]] = []
39    with path.open("r", encoding=encoding) as f:
40        for line in f:
41            line = line.rstrip("\n")
42            if not line.strip():
43                continue
44            urn, _, text = line.partition("|")
45            if urn_filter is not None and urn_filter not in urn:
46                continue
47            pairs.append((urn, text))
48    return pairs
